changes = []
visited = 0


def replace_mtimes(value, prior_value, path="$"):
    global visited
    if isinstance(value, dict) and isinstance(prior_value, dict):
        result = {}
        for key, item in value.items():
            child_path = f"{path}.{key}"
            if key == "mtime" and key in prior_value:
                visited += 1
                replacement = prior_value[key]
                if item != replacement:
                    changes.append((child_path, item, replacement))
                result[key] = replacement
            elif key in prior_value:
                result[key] = replace_mtimes(
                    item, prior_value[key], child_path
                )
            else:
                result[key] = item
        return result
    if isinstance(value, list) and isinstance(prior_value, list):
        return [
            replace_mtimes(item, prior_value[index], f"{path}[{index}]")
            if index < len(prior_value)
            else item
            for index, item in enumerate(value)
        ]
    return value


reconstructed = replace_mtimes(current, prior)
payload = json.dumps(reconstructed).encode()
print("current", hashlib.sha256(current_path.read_bytes()).hexdigest(),
      current_path.stat().st_size)
print("prior-mtime source",
      hashlib.sha256(prior_mtime_path.read_bytes()).hexdigest(),
      prior_mtime_path.stat().st_size)
print("mtime fields", visited, "changed", len(changes))
print("reconstructed", hashlib.sha256(payload).hexdigest(), len(payload))
